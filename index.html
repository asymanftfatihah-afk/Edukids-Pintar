class KidDashboard extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: Color(0xFFFEF3C7),
      body: GridView.count(
        crossCrossCount: 2,
        padding: EdgeInsets.all(16),
        children: [
          _buildKidCard("🔤", "Belajar ABC", Color(0xFF38BDF8)),
          _buildKidCard("🔢", "Nombor", Color(0xFF4ADE80)),
        ],
      ),
    );
  }

  Widget _buildKidCard(String icon, String title, Color color) {
    return Card(
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(32)),
      elevation: 4,
      child: Column(
        mainAxisAlignment: MainAxisAlignment.center,
        children: [
          Text(icon, style: TextStyle(fontSize: 40)),
          Text(title, style: TextStyle(fontWeight: FontWeight.bold, fontSize: 18)),
        ],
      ),
    );
  }
}

4. KOD BACKEND FIREBASE (Cloud Functions - Simpan Ganjaran)

const functions = require('firebase-functions');
const admin = require('firebase-admin');
admin.initializeApp();

exports.tambahBintangMurid = functions.https.onCall(async (data, context) => {
  const muridId = data.muridId;
  const jumlahBintang = data.bintang;
  
  const muridRef = admin.firestore().collection('murid').doc(muridId);
  await muridRef.update({
    'ganjaran.bintang': admin.firestore.FieldValue.increment(jumlahBintang)
  });
  
  return { success: true, message: "Bintang berjaya dikemas kini di cloud!" };
});
